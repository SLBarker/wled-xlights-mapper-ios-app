  <action>
Add the hamburger nav JavaScript. Place it as a NEW `<script>` block immediately AFTER the existing IntersectionObserver `<script>` block, and BEFORE the closing `</body>` tag.

The script must implement:
1. Toggle open/close on button click
2. Close on any nav link click
3. Close on Escape key press
4. Focus management (first link on open, toggle button on close via Escape)
5. aria-label and aria-expanded updates on the toggle button

```html
<script>
  /* ── Hamburger nav toggle ── */
  (function () {
    var toggle = document.querySelector('.site-nav__toggle');
    var nav    = document.querySelector('.site-nav');
    var drawer = document.getElementById('site-nav-drawer');

    if (!toggle || !drawer) return;

    var firstLink = drawer.querySelector('a');

    function openMenu() {
      nav.classList.add('site-nav--open');
      drawer.classList.add('site-nav__drawer--open');
      drawer.removeAttribute('aria-hidden');
      toggle.setAttribute('aria-expanded', 'true');
      toggle.setAttribute('aria-label', 'Close navigation menu');
      if (firstLink) firstLink.focus();
    }

    function closeMenu(returnFocus) {
      nav.classList.remove('site-nav--open');
      drawer.classList.remove('site-nav__drawer--open');
      drawer.setAttribute('aria-hidden', 'true');
      toggle.setAttribute('aria-expanded', 'false');
      toggle.setAttribute('aria-label', 'Open navigation menu');
      if (returnFocus) toggle.focus();
    }

    toggle.addEventListener('click', function () {
      if (nav.classList.contains('site-nav--open')) {
        closeMenu(false);
      } else {
        openMenu();
      }
    });

    /* Close on any drawer link tap */
    drawer.querySelectorAll('a').forEach(function (link) {
      link.addEventListener('click', function () {
        closeMenu(false);
      });
    });

    /* Close on Escape */
    document.addEventListener('keydown', function (e) {
      if (e.key === 'Escape' && nav.classList.contains('site-nav--open')) {
        closeMenu(true);
      }
    });
  }());
</script>
```

This must be placed verbatim after the existing `</script>` closing tag of the IntersectionObserver block. Do not modify the existing IntersectionObserver script.
